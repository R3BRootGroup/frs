87 void FRSMusicMapped2Cal::SetParameter()
88 {
89 
90  //--- Parameter Container ---
91  NumDets = fCal_Par->GetNumDets(); // Number of Detectors
92  NumAnodes = fCal_Par->GetNumAnodes(); // Number of anodes
93  NumParams = fCal_Par->GetNumParametersFit(); // Number of Parameters
94 
95  LOG(info) << "FRSMusicMapped2Cal: Nb detectors: " << NumDets;
96  LOG(info) << "FRSMusicMapped2Cal: Nb anodes: " << NumAnodes;
97  LOG(info) << "FRSMusicMapped2Cal: Nb parameters from pedestal fit: " << NumParams;
98 
99  CalParams = new TArrayF();
100  Int_t array_size = NumDets * NumAnodes * NumParams;
101  CalParams->Set(array_size);
102  CalParams = fCal_Par->GetAnodeCalParams(); // Array with the Cal parameters
103 
104  // Count the number of dead anodes per Music detector
105  for (Int_t d = 0; d < NumDets; d++)
106  {
107  Int_t numdeadanodes = 0;
108  for (Int_t i = 0; i < NumAnodes; i++)
109  if (CalParams->GetAt(NumParams * i + 1 + NumAnodes * d * NumParams) == -1)
110  numdeadanodes++;
111  LOG(info) << "FRSMusicMapped2Cal: Nb of dead anodes in MUSIC " << d + 1 << ": " << numdeadanodes;
112  }
113 }

156 // ----- Public method Execution --------------------------------------------
157 void FRSMusicMapped2Cal::Exec(Option_t* option)
158 {
159  // Reset entries in output arrays, local arrays
160  Reset();
161 
162  if (!fCal_Par)
163  {
164  LOG(error) << "NO Container Parameter!!";
165  }
166 
167  // Reading the Input -- Mapped Data --
168  Int_t nHits = fMusicMappedDataCA->GetEntries();
169  if (nHits != NumAnodes * NumDets && nHits > 0)
170  LOG(warn) << "FRSMusicMapped2Cal: nHits!=" << nHits << " NumAnodes:NumDets" << NumAnodes << ":" << NumDets;
171  if (!nHits)
172  return;
173 
174  FRSMusicMappedData** mappedData = new FRSMusicMappedData*[nHits];
175  Int_t detId;
176  Int_t anodeId;
177  Double_t energy;
178  Double_t pedestal = 0.;
179  Double_t sigma = 0.;
180 
181  for (Int_t i = 0; i < nHits; i++)
182  {
183  mappedData[i] = (FRSMusicMappedData*)(fMusicMappedDataCA->At(i));
184  detId = mappedData[i]->GetDetectorId();
185  anodeId = mappedData[i]->GetAnodeId();
186 
187  pedestal = CalParams->GetAt(NumParams * anodeId + 1 + detId * NumAnodes * NumParams);
188  // sigma=CalParams->GetAt(NumParams*anodeId+2+detId*NumAnodes*NumParams);
189 
190  // std::cout << detId << " " << anodeId<<" "<< mappedData[i]->GetEnergy()<< " " << pedestal << std::endl;
191 
192  energy = mappedData[i]->GetEnergy() - pedestal;
193 
194  if (pedestal != -1)
195  AddCalData(detId, anodeId, energy);
196  }
197 
198  if (mappedData)
199  delete mappedData;
200  return;
201 }
