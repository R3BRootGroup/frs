150 // ----- Public method Execution --------------------------------------------
151 void FRSMusicCal2Hit::Exec(Option_t* option)
152 {
153  // Reset entries in output arrays, local arrays
154  Reset();
155 
156  if (!fCal_Par)
157  {
158  LOG(error) << "NO Container Parameter!!";
159  }
160 
161  Int_t nHits = fMusicCalDataCA->GetEntries();
162  if (!nHits)
163  return;
164 
165  // FRSMusicCalData* CalDat;
166  FRSMusicCalData** CalDat = new FRSMusicCalData*[nHits];
167 
168  Int_t detId, anodeId;
169  Double_t energyperanode[5][8]; // max 5 detectors and 8 anodes
170  Int_t nbdet = 0;
171 
172  for (Int_t i = 0; i < 5; i++)
173  for (Int_t j = 0; j < 8; j++)
174  energyperanode[i][j] = 0;
175 
176  for (Int_t i = 0; i < nHits; i++)
177  {
178  CalDat[i] = (FRSMusicCalData*)(fMusicCalDataCA->At(i));
179  detId = CalDat[i]->GetDetectorId();
180  anodeId = CalDat[i]->GetAnodeId();
181 
182  energyperanode[detId][anodeId] = CalDat[i]->GetEnergy();
183  nbdet = detId;
184  }
185 
186  Double_t a0, a1;
187 
188  // calculate truncated dE from 8 anodes, Munich MUSIC
189  for (Int_t i = 0; i <= NumDets; i++)
190  {
191 
192  Float_t r1 = sqrt(energyperanode[i][0] * energyperanode[i][1]);
193  Float_t r2 = sqrt(energyperanode[i][2] * energyperanode[i][3]);
194  Float_t r3 = sqrt(energyperanode[i][4] * energyperanode[i][5]);
195  Float_t r4 = sqrt(energyperanode[i][6] * energyperanode[i][7]);
196 
197  if ((r1 > 0) && (r2 > 0) && (r3 > 0) && (r4 > 0))
198  {
199  a0 = CalParams->GetAt(i * NumParams);
200  a1 = CalParams->GetAt(i * NumParams + 1);
201  AddHitData(i, a0 + a1 * sqrt(sqrt(sqrt(r1 * r2) * sqrt(r3 * r4))));
202  }
203  }
204 
205  if (CalDat)
206  delete CalDat;
207  return;
208 }
