159 // ----- Public method Exec --------------------------------------------
160 void R3BFrsHit2AnaS4Par::Exec(Option_t* opt)
161 {
162 
163  Double_t fZ = 0., fAq = 0.;
164 
165  Int_t nHitFrs = fFrsMappedDataCA->GetEntries();
166  Int_t nHitMusic = fMusicHitDataCA->GetEntries();
167  Int_t nHitTpc = fTpcHitDataCA->GetEntries();
168  // LOG(info) << nHitMusic << " " << nHitTpc ;
169  if (!nHitMusic || !nHitFrs || nHitTpc < 4)
170  return; // FIXME:include here warn!
171 
172  R3BFrsMappedData** MapFrs = new R3BFrsMappedData*[nHitFrs];
173  FRSMusicHitData** HitMusic = new FRSMusicHitData*[nHitMusic];
174  R3BTpcHitData** HitTpc = new R3BTpcHitData*[nHitTpc];
175 
176  // Z from musics ------------------------------------
177  Double_t countz = 0;
178  for (Int_t i = 0; i < nHitMusic; i++)
179  {
180  HitMusic[i] = (FRSMusicHitData*)(fMusicHitDataCA->At(i));
181  if (HitMusic[i]->GetZ() > 1)
182  {
183  fZ = fZ + HitMusic[i]->GetZ();
184  countz++;
185  }
186  }
187  if (countz > 0)
188  {
189  fZ = fZ / countz;
190  }
191  else
192  {
193  fZ = 0.;
194  }
195 
196  // Positions from TPCs ------------------------------
197  Double_t tpc_x[4];
198  Int_t detID = 0;
199  for (Int_t i = 0; i < nHitTpc; i++)
200  {
201  HitTpc[i] = (R3BTpcHitData*)(fTpcHitDataCA->At(i));
202  detID = HitTpc[i]->GetDetectorId();
203  tpc_x[detID] = HitTpc[i]->GetX();
204  }
205 
206  Int_t SCI24_TofRR = 0.;
207  Int_t SCI24_TofLL = 0.;
208 
209  for (Int_t i = 0; i < nHitFrs; i++)
210  {
211  MapFrs[i] = (R3BFrsMappedData*)(fFrsMappedDataCA->At(i));
212  SCI24_TofRR = MapFrs[i]->GetSCI41RT();
213  SCI24_TofLL = MapFrs[i]->GetSCI41LT();
214  // LOG(info) << SCI24_TofRR << " " << SCI24_TofLL ;
215  }
216 
217  // FOCAL POSITION S2
218  double angle_S2_mrad = (tpc_x[1] - tpc_x[0]) / fDisTpcS2 * 1000.;
219  double x_position_focal_S2 = (tpc_x[1] + fPosFocalS2 * tan(angle_S2_mrad / 1000.));
220 
221  // FOCAL POSITION S4
222  double angle_S4_mrad = (tpc_x[3] - tpc_x[2]) / fDisTpcS4 * 1000.;
223  double x_position_focal_S4 = tpc_x[2] + fPosFocalS4 * tan(angle_S4_mrad / 1000.);
224 
225  // Velocity S2-S4
226  double TAC_CAL_SC24_LL = -0.01045;
227  double TAC_CAL_SC24_RR = -0.01095;
228  double ToF_star_S2_S4 = 0.5 * (TAC_CAL_SC24_LL * SCI24_TofLL + TAC_CAL_SC24_RR * SCI24_TofRR);
229  double Beta_S2_S4 = ((fPathS2S4 / (fTOFS2S4 + ToF_star_S2_S4)) * pow(10., 7)) / c;
230  double Gamma_S2_S4 = 1. / (sqrt(1. - (Beta_S2_S4) * (Beta_S2_S4)));
231 
232  // Brho and A/q
233  double Brho_S4 = fBfield_S2_S4 * frho_S2_S4 *
234  (1. - (((x_position_focal_S4 / 1000.) - fMagS2S4 * (x_position_focal_S2 / 1000.)) / fDispS2S4));
235  fAq = (Brho_S4 * e) / (Gamma_S2_S4 * Beta_S2_S4 * c * u);
236 
237  // Fill histograms for calibrations
238  if (fZ > fCutZ - 0.5 && fZ < fCutZ + 0.5)
239  {
240  fh_anglevsAq->Fill(angle_S4_mrad, fAq);
241  graph->SetPoint(fNb, angle_S4_mrad, fAq);
242  fNb++;
243  }
244 
245  if (HitMusic)
246  delete HitMusic;
247  if (HitTpc)
248  delete HitTpc;
249  return;
250 }
