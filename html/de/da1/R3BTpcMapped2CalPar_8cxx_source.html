327 void R3BTpcMapped2CalPar::SearchCalParXY()
328 {
329 
330  LOG(info) << "R3BTpcMapped2CalPar: Search calibration parameters";
331 
332  Int_t numPars = 2; // by default number of parameters=2
333  fTpc_Par->SetNumDets(fNumDets);
334 
335  Int_t fNumAnodesTotal = fNumAnodesX + fNumAnodesY;
336  fTpc_Par->SetNumDT(fNumAnodesTotal);
337  fTpc_Par->SetNumParametersFit(numPars);
338  fTpc_Par->GetTpcCalParams()->Set(numPars * fNumAnodesTotal * fNumDets);
339 
340  int fNumPeaks = 8;
341 
342  for (Int_t d = 0; d < fNumDets; d++)
343  {
344 
345  for (Int_t i = 0; i < fNumAnodesTotal; i++)
346  {
347 
348  TSpectrum* ss = new TSpectrum(fNumPeaks);
349  Int_t nfound = 0;
350 
351  // Fit of DT in the Y direction
352  if (i < fNumAnodesY)
353  {
354  if (fh_Map_dt_y[i + d * fNumAnodesY]->GetEntries() > fMinStatistics)
355  {
356  // Sigma=1 is better than other values!
357  nfound = ss->Search(fh_Map_dt_y[i + d * fNumAnodesY], 1, "", 0.00001);
358 
359  Int_t idx[nfound];
360  Int_t idy[nfound];
361 
362  Double_t* xpeaks = ss->GetPositionX();
363  TMath::Sort(nfound, xpeaks, idx, kTRUE);
364 
365  Double_t* ypeaks = ss->GetPositionY();
366  TMath::Sort(nfound, ypeaks, idy, kTRUE);
367  double ypos[nfound];
368  SearchMax(xpeaks, ypeaks, nfound, ypos);
369 
370  // Calibrated Spectrum
371  Double_t X[nfound];
372  Double_t Y[nfound];
373 
374  if (nfound < 3)
375  LOG(error) << "R3BTpccMapped2CalPar::SearchY() Couldn't get the sufficient parameters:"
376  << nfound << "<2";
377 
378  // LOG(info)<<"nfound "<<nfound<<FairLogger::endl;
379 
380  nfound = TMath::Min(nfound, 3);
381 
382  for (Int_t j = 0; j < nfound; j++)
383  {
384  X[j] = ypos[j];
385  Y[j] = fDeltaY * j - fDeltaY;

387  }
388 
389  TF1* f1 = new TF1("f1", "[0]+[1]*x", fMapHistos_left, fMapHistos_right);
390  TGraph* graph = new TGraph(nfound, X, Y);
391  graph->Fit("f1", "QR"); // Quiet mode (minimum printing)
392 
393  if (nfound > 1)
394  {
395  for (Int_t h = 0; h < numPars; h++)
396  fTpc_Par->SetTpcCalParams(f1->GetParameter(h),
397  numPars * i + h + d * numPars * fNumAnodesTotal);
398  }
399  else
400  {
401  fTpc_Par->SetTpcCalParams(0, numPars * i + d * numPars * fNumAnodesTotal);
402  fTpc_Par->SetTpcCalParams(1, numPars * i + d * numPars * fNumAnodesTotal + 1);
403  LOG(info) << "R3BTpccMapped2CalPar::SearchY() default parameters: a0=0 and a1=1";
404  }
405  }
406  else
407  {
408  fTpc_Par->SetTpcCalParams(0, numPars * i + d * numPars * fNumAnodesTotal); // dead anode
409  fTpc_Par->SetTpcCalParams(0, numPars * i + d * numPars * fNumAnodesTotal + 1);

412  }
413  }

416
