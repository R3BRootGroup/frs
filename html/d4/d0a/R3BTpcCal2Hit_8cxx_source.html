128 // ----- Public method Execution --------------------------------------------
129 void R3BTpcCal2Hit::Exec(Option_t* option)
130 {
131 
132  // Reset entries in output arrays, local arrays
133  Reset();
134 
135  if (!fCal_Par)
136  {
137  LOG(error) << "NO Container Parameter!!";
138  }
139 
140  Int_t nHits = fTpcCalDataCA->GetEntries();
141  if (!nHits)
142  return;
143 
144  R3BTpcCalData** CalDat = new R3BTpcCalData*[nHits];
145 
146  Int_t detId, secId, xyId;
147  Double_t fx = -999., fy = -999.;
148 
149  Double_t tpc_x[fNumDets], tpc_y[fNumDets];
150  Int_t count_x[fNumDets], count_y[fNumDets];
151  for (Int_t i = 0; i < fNumDets; i++)
152  {
153  tpc_x[i] = -500.;
154  count_x[i] = 0;
155  tpc_y[i] = -500.;
156  count_y[i] = 0;
157  }
158 
159  for (Int_t i = 0; i < nHits; i++)
160  {
161  CalDat[i] = (R3BTpcCalData*)(fTpcCalDataCA->At(i));
162  detId = CalDat[i]->GetDetectorId();
163  secId = CalDat[i]->GetSecId();
164  xyId = CalDat[i]->GetXYId();
165 
166  // std::cout << detId << " " << secId << " " << xyId << std::endl;
167 
168  if (xyId == 0)
169  { // for X
170  if (CalDat[i]->GetPosition() >= -100. && CalDat[i]->GetPosition() <= 100.)
171  {
172  if (tpc_x[detId] == -500)
173  tpc_x[detId] = 0.;
174  tpc_x[detId] = tpc_x[detId] + CalDat[i]->GetPosition();
175  count_x[detId]++;
176  }
177  }
178  else
179  { // for Y
180  if (CalDat[i]->GetPosition() >= -100. && CalDat[i]->GetPosition() <= 100.)
181  {
182  if (tpc_y[detId] == -500)
183  tpc_y[detId] = 0.;
184  tpc_y[detId] = tpc_y[detId] + CalDat[i]->GetPosition();
185  count_y[detId]++;
186  }
187  }
188  }
189 
190  for (Int_t i = 0; i < fNumDets; i++)
191  {
192  if (count_x[i] > 0) // && count_y[i] > 0)
193  {
194  fx = tpc_x[i] / count_x[i];
195  if (count_y[i] > 0)
196  fy = tpc_y[i] / count_y[i];
197  // std::cout << i << " " << fx << " " << fy << std::endl;
198  AddHitData(i, fx, fy);
199  }
200  }
201 
202  if (CalDat)
203  delete CalDat;
204  return;
205 }
