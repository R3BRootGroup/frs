72 // ----- Public method ProcessHits --------------------------------------
73 Bool_t WASATof::ProcessHits(FairVolume* vol)
74 {
75  // Simple Det PLane
76  if (gMC->IsTrackEntering())
77  {
78  fELoss = 0.;
79  fTime_in = gMC->TrackTime() * 1.0e09;
80  fLength_in = gMC->TrackLength();
81  gMC->TrackPosition(fPosIn);
82  gMC->TrackMomentum(fMomIn);
83  }
84 
85  // Sum energy loss for all steps in the active volume
86  fELoss += gMC->Edep();
87 
88  // Set additional parameters at exit of active volume. Create WASATofPoint.
89  if (gMC->IsTrackExiting() || gMC->IsTrackStop() || gMC->IsTrackDisappeared())
90  {
91  fTrackID = gMC->GetStack()->GetCurrentTrackNumber();
92  fVolumeID = vol->getMCid();
93  fDetCopyID = vol->getCopyNo();
94  gMC->TrackPosition(fPosOut);
95  gMC->TrackMomentum(fMomOut);
96  if (fELoss == 0.)
97  return kFALSE;
98 
99  fTime_out = gMC->TrackTime() * 1.0e09; // also in case particle is stopped in detector, or decays...
100  fLength_out = gMC->TrackLength();
101  fTime = (fTime_out + fTime_in) / 2.;
102  fLength = (fLength_out + fLength_in) / 2.;
103 
104  if (gMC->IsTrackExiting())
105  {
106  const Double_t* oldpos;
107  const Double_t* olddirection;
108  Double_t newpos[3];
109  Double_t newdirection[3];
110  Double_t safety;
111 
112  gGeoManager->FindNode(fPosOut.X(), fPosOut.Y(), fPosOut.Z());
113  oldpos = gGeoManager->GetCurrentPoint();
114  olddirection = gGeoManager->GetCurrentDirection();
115 
116  for (Int_t i = 0; i < 3; i++)
117  {
118  newdirection[i] = -1 * olddirection[i];
119  }
120 
121  gGeoManager->SetCurrentDirection(newdirection);
122  safety = gGeoManager->GetSafeDistance();
123 
124  gGeoManager->SetCurrentDirection(-newdirection[0], -newdirection[1], -newdirection[2]);
125 
126  for (Int_t i = 0; i < 3; i++)
127  {
128  newpos[i] = oldpos[i] - (3 * safety * olddirection[i]);
129  }
130 
131  fPosOut.SetX(newpos[0]);
132  fPosOut.SetY(newpos[1]);
133  fPosOut.SetZ(newpos[2]);
134  }
135 
136  AddHit(fTrackID,
137  fVolumeID,
138  fDetCopyID,
139  TVector3(fPosIn.X(), fPosIn.Y(), fPosIn.Z()),
140  TVector3(fPosOut.X(), fPosOut.Y(), fPosOut.Z()),
141  TVector3(fMomIn.Px(), fMomIn.Py(), fMomIn.Pz()),
142  TVector3(fMomOut.Px(), fMomOut.Py(), fMomOut.Pz()),
143  fTime,
144  fLength,
145  fELoss,
146  gMC->TrackPid());
147 
148  // Increment number of TofPoints for this track
149  R3BStack* stack = (R3BStack*)gMC->GetStack();
150  stack->AddPoint(kTOF);
151 
152  ResetParameters();
153  }
154 
155  return kTRUE;
156 }

202 // ----- Public method CopyClones -----------------------------------------
203 void WASATof::CopyClones(TClonesArray* cl1, TClonesArray* cl2, Int_t offset)
204 {
205  Int_t nEntries = cl1->GetEntriesFast();
206  LOG(info) << "WASATof: " << nEntries << " entries to add";
207  TClonesArray& clref = *cl2;
208  WASATofPoint* oldpoint = NULL;
209  for (Int_t i = 0; i < nEntries; i++)
210  {
211  oldpoint = (WASATofPoint*)cl1->At(i);
212  Int_t index = oldpoint->GetTrackID() + offset;
213  oldpoint->SetTrackID(index);
214  new (clref[fPosIndex]) WASATofPoint(*oldpoint);
215  fPosIndex++;
216  }
217  LOG(info) << "WASATof: " << cl2->GetEntriesFast() << " merged entries";
218 }
