200 // ----- Public method Execution --------------------------------------------
201 void R3BFrsHit2AnaS4::Exec(Option_t* option)
202 {
203 
204  // if(++nEvents % 10000 == 0)
205  // LOG(info) << nEvents ;
206 
207  // Reset entries in output arrays, local arrays
208  Reset();
209 
210  Double_t fZ = 0., fAq = 0.;
211 
212  Int_t nHitFrs = fFrsMappedDataCA->GetEntries();
213  Int_t nHitMusic = fMusicHitDataCA->GetEntries();
214  Int_t nHitTpc = fTpcHitDataCA->GetEntries();
215  // LOG(info) << nHitMusic << " " << nHitTpc ;
216  if (!nHitMusic || !nHitFrs || !nHitTpc)
217  return; // FIXME:include here warn!
218 
219  // R3BFrsMappedData** MapFrs = new R3BFrsMappedData*[nHitFrs];
220  FrsSciSingleTcalData** MapFrs = new FrsSciSingleTcalData*[nHitFrs];
221  FRSMusicHitData** HitMusic = new FRSMusicHitData*[nHitMusic];
222  R3BTpcHitData** HitTpc = new R3BTpcHitData*[nHitTpc];
223 
224  // Z from musics ------------------------------------
225  Double_t countz = 0;
226  for (Int_t i = 0; i < nHitMusic; i++)
227  {
228  HitMusic[i] = (FRSMusicHitData*)(fMusicHitDataCA->At(i));
229  if (HitMusic[i]->GetZ() > 1)
230  {
231  fZ = fZ + HitMusic[i]->GetZ();
232  countz++;
233  }
234  }
235  if (countz > 0)
236  {
237  fZ = fZ / countz;
238  }
239  else
240  {
241  fZ = 0.;
242  }
243 
244  // Positions from TPCs ------------------------------
245  Double_t tpc_x[4];
246  Int_t detID = 0;
247  for (Int_t i = 0; i < nHitTpc; i++)
248  {
249  HitTpc[i] = (R3BTpcHitData*)(fTpcHitDataCA->At(i));
250  detID = HitTpc[i]->GetDetectorId();
251  tpc_x[detID] = HitTpc[i]->GetX();
252  }
253 
254  Double_t SCI24_TofRR = 0.;
255  Double_t SCI24_TofLL = 0.;
256 
257  for (Int_t i = 0; i < nHitFrs; i++)
258  {
259  MapFrs[i] = (FrsSciSingleTcalData*)(fFrsMappedDataCA->At(i));
260  SCI24_TofRR = MapFrs[i]->GetRawTofNs(0);
261  SCI24_TofLL = MapFrs[i]->GetRawTofNs(0);
262  // SCI24_TofRR = MapFrs[i]->GetSCI41RT();
263  // SCI24_TofLL = MapFrs[i]->GetSCI41LT();
264  // LOG(info) << SCI24_TofRR << " " << SCI24_TofLL ;
265  }
266 
267  // FOCAL POSITION S2
268  double angle_S2_mrad = (tpc_x[1] - tpc_x[0]) / fDisTpcS2 * 1000.;
269  double x_position_focal_S2 = (tpc_x[1] + fPosFocalS2 * tan(angle_S2_mrad / 1000.));
270 
271  // FOCAL POSITION S4
272  double angle_S4_mrad = (tpc_x[3] - tpc_x[2]) / fDisTpcS4 * 1000.;
273  double x_position_focal_S4 = tpc_x[2] + fPosFocalS4 * tan(angle_S4_mrad / 1000.);
274 
275  // Velocity S2-S4
276  double TAC_CAL_SC24_LL = -0.01045;
277  double TAC_CAL_SC24_RR = -0.01095;
278  double ToF_star_S2_S4 = 0.5 * (SCI24_TofLL + SCI24_TofRR);
279  double Beta_S2_S4 = ((fPathS2S4 / (fTOFS2S4 + ToF_star_S2_S4)) * pow(10., 7)) / c;
280  double Gamma_S2_S4 = 1. / (sqrt(1. - (Beta_S2_S4) * (Beta_S2_S4)));
281 
282  // Brho and A/q
283  double Brho_S4 = fBfield_S2_S4 * frho_S2_S4 *
284  (1. - (((x_position_focal_S4 / 1000.) - fMagS2S4 * (x_position_focal_S2 / 1000.)) / fDispS2S4));
285  fAq = (Brho_S4 * e) / (Gamma_S2_S4 * Beta_S2_S4 * c * u);
286 
287  // Correction for dependence on angle
288  fAq = (fAq - fParm1) * cos(-1. * fRotS4) + (angle_S4_mrad - fParm0) * sin(-1. * fRotS4) + fParm1;
289 
290  // Fill the data
291  AddData(fZ + fOffsetZ, fAq + fOffsetAq, x_position_focal_S2, angle_S2_mrad, x_position_focal_S4, Beta_S2_S4);
292 
293  if (HitMusic)
294  delete HitMusic;
295  if (HitTpc)
296  delete HitTpc;
297  if (MapFrs)
298  delete MapFrs;
299  return;
300 }
