75 void FRSResonanceGenerator::ProjectileExcitation(FairPrimaryGenerator* primGen,
76  Double_t mres,
77  Double_t wres,
78  Double_t mnucleon)
79 {
80  Double_t MD = gRandom->BreitWigner(mres, wres);
81  auto mom = DecayCM(MD, mnucleon, masspion);
82  lvpion1.SetPxPyPzE(mom.X(), mom.Y(), mom.Z(), TMath::Sqrt(mom.Mag2() + masspion * masspion));
83  lvpion1.Boost(fBeta);
84  primGen->AddTrack(-211, lvpion1.X(), lvpion1.Y(), lvpion1.Z(), fvx, fvy, fvz);
85 
86  lvn.SetPxPyPzE(-mom.X(), -mom.Y(), -mom.Z(), TMath::Sqrt(mom.Mag2() + mnucleon * mnucleon));
87  lvn.Boost(fBeta);
88  if (mnucleon == massproton)

151 TVector3 FRSResonanceGenerator::DecayCM(Double_t M, Double_t m1, Double_t m2)
152 {
153  TVector3 p;
154  Double_t px, py, pz;
155  p.SetXYZ(0., 0., 0.);
156  if (M < m1 + m2)
157  M = m1 + m2 + 0.00001;
158  Double_t mom = sqrt((M * M - (m1 + m2) * (m1 + m2)) * (M * M - (m1 - m2) * (m1 - m2))) / 2. / M;
159  auto rand = new TRandom3;
160  rand->Sphere(px, py, pz, 1.0);
161  p.SetXYZ(mom * px, mom * py, mom * pz);
162  return p;
163 }
