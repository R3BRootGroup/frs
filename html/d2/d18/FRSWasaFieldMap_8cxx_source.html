121 Bool_t FRSWasaFieldMap::IsInside(Double_t x, Double_t y, Double_t z)
122 {
123  Double_t rl = TMath::Sqrt(x * x + y * y);
124 
125  // --- Check for being outside the map range
126  if (rl >= fRmin && rl <= fRmax && z >= fZmin && z <= fZmax)
127  {
128  return kTRUE;
129  }
130 
131  return kFALSE;
