128 // ----- Public method Exec --------------------------------------------
129 void FRSMusicCal2HitPar::Exec(Option_t* opt)
130 {
131 
132  Int_t nHits = fMusicCalDataCA->GetEntries();
133  if (!nHits)
134  return;
135 
136  FRSMusicCalData* CalDat;
137  Int_t detId, anodeId;
138  Double_t energyperanode[5][8]; // max 5 detectors and 8 anodes
139  Int_t nbdet = 0;
140 
141  for (Int_t i = 0; i < 5; i++)
142  for (Int_t j = 0; j < 8; j++)
143  energyperanode[i][j] = 0;
144 
145  for (Int_t i = 0; i < nHits; i++)
146  {
147  CalDat = (FRSMusicCalData*)(fMusicCalDataCA->At(i));
148  detId = CalDat->GetDetectorId();
149  anodeId = CalDat->GetAnodeId();
150 
151  energyperanode[detId][anodeId] = CalDat->GetEnergy();
152  nbdet = detId;
153  }
154 
155  /* calculate truncated dE from 8 anodes, Munich MUSIC */
156  for (Int_t i = 0; i <= nbdet; i++)
157  {
158 
159  Float_t r1 = sqrt(energyperanode[i][0] * energyperanode[i][1]);
160  Float_t r2 = sqrt(energyperanode[i][2] * energyperanode[i][3]);
161  Float_t r3 = sqrt(energyperanode[i][4] * energyperanode[i][5]);
162  Float_t r4 = sqrt(energyperanode[i][6] * energyperanode[i][7]);
163 
164  if ((r1 > 0) && (r2 > 0) && (r3 > 0) && (r4 > 0))
165  {
166  // Fill Histos
167  Double_t charge = sqrt(sqrt(sqrt(r1 * r2) * sqrt(r3 * r4)));
168  if (charge > 0.)
169  fh_Cal_energy_det[i]->Fill(charge);
170  // std::cout << i << " z = " << charge << std::endl;
171  }
172  }
173 }

192 void FRSMusicCal2HitPar::SearchZ()
193 {
194 
195  LOG(info) << "FRSMusicCal2HitPar: Search atomic numbers (Z)";
196 
197  Int_t numPars = 2; // by default number of parameters=3
198 
199  fDet_Par->SetNumDets(fNumDets);
200  fDet_Par->GetDetectorHitParams()->Set(numPars * fNumDets);
201 
202  int fNumPeaks = 8;
203  TSpectrum* ss = new TSpectrum(fNumPeaks);
204 
205  for (Int_t i = 0; i < fNumDets; i++)
206  {
207 
208  Int_t nfound = 0;
209 
210  if (fh_Cal_energy_det[i]->GetEntries() > fMinStatistics)
211  {
212  // Sigma=1 is better than other values!
213  nfound = ss->Search(fh_Cal_energy_det[i], 1, "", 0.1);
214 
215  Int_t idx[nfound];
216  Int_t idy[nfound];
217 
218  Double_t* xpeaks = ss->GetPositionX();
219  // Double_t *ypeaks = ss->GetPositionY();
220  TMath::Sort(nfound, xpeaks, idx, kTRUE);
221  // TMath::Sort(nfound, ypeaks, idy, kTRUE);
222 
223  // Calibrated Spectrum
224  Double_t X[nfound];
225  Double_t Y[nfound];
226 
227  if (nfound < 2)
228  LOG(error) << "FRSMusicCal2HitPar::SearchZ() Couldn't get the sufficient parameters:" << nfound << "<2";
229 
230  nfound = TMath::Min(nfound, fNumPeaks);
231 
232  for (Int_t j = 0; j < nfound; j++)
233  {
234  // if(ypeaks[idy[j]]>5.){
235  X[j] = xpeaks[idx[j]];
236  Y[j] = fMaxZ - j;
237  // }
238  // std::cout << X[j] << " " << Y[j] << std::endl;
239  }
240 
241  // X[nfound]=0.;
242  // Y[nfound]=0.;
243 
244  TF1* f1 = new TF1("f1", "[0]+[1]*x", fMapHistos_left, fMapHistos_right);
245  TGraph* graph = new TGraph(nfound, X, Y);
246  graph->Fit("f1", "QR"); // Quiet mode (minimum printing)
247 
248  if (nfound > 1)
249  {
250  for (Int_t h = 0; h < numPars; h++)
251  fDet_Par->SetDetectorHitParams(f1->GetParameter(h), numPars * i + h);
252  }
253  else
254  {
255  fDet_Par->SetDetectorHitParams(0, numPars * i);
256  fDet_Par->SetDetectorHitParams(1, numPars * i + 1);
257  LOG(info) << "FRSMusicCal2HitPar::SearchZ() default parameters: a0=0 and a1=1";
258  }
259  }
260  }
261 
262  delete ss;
263  fDet_Par->setChanged();
264  return;
265 }
