215 void FrsSciMapped2TcalPar::CalculateVftxTcalParams()
216 {
217  LOG(info) << "FrsSciMapped2TcalPar: CalculateVftxTcalParams()";
218 
219  UInt_t IntegralTot;
220  UInt_t IntegralPartial;
221  Double_t Bin2Ns[fNumTcalParsPerSignal];
222 
223  for (Int_t sig = 0; sig < fNumSignals; sig++)
224  {
225  if (fh_TimeFineBin[sig]->GetEntries() > fMinStatistics)
226  {
227  IntegralTot = fh_TimeFineBin[sig]->Integral();
228  IntegralPartial = 0;
229  for (Int_t bin = 0; bin < fNumTcalParsPerSignal; bin++)
230  {
231  IntegralPartial += fh_TimeFineBin[sig]->GetBinContent(bin + 1);
232  Bin2Ns[bin] = 5. * ((Double_t)IntegralPartial) / (Double_t)IntegralTot;
233  fh_TimeFineNs[sig]->SetBinContent(bin + 1, Bin2Ns[bin]);
234  fTcalPar->SetSignalTcalParams(Bin2Ns[bin], sig * fNumTcalParsPerSignal + bin);
235  }
236  }
237  fh_TimeFineNs[sig]->Write(); // empty histo if stat <fMinStatistics
238  fh_TimeFineBin[sig]->Write();
239  }
240  fTcalPar->setChanged();
241  return;
242 }
