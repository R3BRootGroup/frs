163 void FRSMusicMapped2CalPar::SearchPedestals()
164 {
165 
166  LOG(info) << "FRSMusicMapped2CalPar: Search pedestals";
167 
168  Int_t numPars = 3; // by default number of parameters=3
169 
170  fAnode_Par->SetNumDets(fNumDets);
171  fAnode_Par->SetNumAnodes(fNumAnodes);
172  fAnode_Par->GetAnodeCalParams()->Set(numPars * fNumAnodes * fNumDets);
173 
174  for (Int_t d = 0; d < fNumDets; d++)
175  {
176  for (Int_t i = 0; i < fNumAnodes; i++)
177  {
178 
179  if (fh_Map_energy_anode[i + d * fNumAnodes]->GetEntries() > fMinStatistics)
180  {
181 
182  TF1* f1 = new TF1("f1", "gaus", fMapHistos_left, fMapHistos_right);
183  f1->SetParameter(1, 100.);
184  f1->SetParameter(2, 2.);
185 
186  fh_Map_energy_anode[i + d * fNumAnodes]->Fit("f1", "RQ0");
187 
188  // Fill container:
189  fAnode_Par->SetAnodeCalParams(f1->GetParameter(0), numPars * i + d * numPars * fNumAnodes);
190  if (f1->GetParameter(2) < fMaxSigma)
191  {
192  fAnode_Par->SetAnodeCalParams(f1->GetParameter(1), numPars * i + d * numPars * fNumAnodes + 1);
193  }
194  else
195  {
196  fAnode_Par->SetAnodeCalParams(-1, numPars * i + d * fNumAnodes + 1); // dead anode
197  }
198  fAnode_Par->SetAnodeCalParams(f1->GetParameter(2), numPars * i + d * numPars * fNumAnodes + 2);
199  }
200  else
201  {
202  fAnode_Par->SetAnodeCalParams(-1, numPars * i + d * numPars * fNumAnodes + 1); // dead anode
203  fAnode_Par->SetAnodeCalParams(0, numPars * i + d * numPars * fNumAnodes + 2);
204  LOG(warn) << "Histogram NO Fitted, detector: " << d + 1 << ", anode: " << i + 1;
205  }
206  }
207  }
208 
209  fAnode_Par->setChanged();
210  return;
211 }
